    """Replace '#...' comments with spaces (keep offsets, newlines, quotes aware)."""
    out = []

        c = s[i]
        if in_str:
            out.append(c)
            if c == '"':
                in_str = False
            i += 1
        elif c == '"':
            in_str = True
            out.append(c)
            i += 1
        elif c == '#':

                i += 1

def strip_comments(s):
    out=[]; i,n=0,len(s); ins=False

            out.append(c)
            if c=='"': ins=False

        elif c=='"':
